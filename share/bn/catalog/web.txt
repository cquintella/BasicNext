# web texts and contract names (bn_catalog: no message is hard-coded).
# One `key = value` per line; `{name}` is filled by the code.

# Failures (bn_rt web_error: message, then cause).
method.unsupported.message = unsupported HTTP method "{method}"
selector.unsupported.message = unsupported selector "{selector}"
cookie.secure_cleartext.message = Secure cookie on a cleartext response
cookie.secure_cleartext.cause = a Secure cookie is sent only over TLS; this request arrived over cleartext HTTP
dns.timeout.message = DNS resolution timed out
dns.pool_limit.message = DNS resolution pool limit exceeded
route.param_missing.message = route parameter not found
route.param_missing.cause = the matched route pattern has no `:name` segment with that name
request.cookie_missing.message = request cookie not found
request.cookie_missing.cause = the request's Cookie headers carry no cookie with that name
session.disabled.message = sessions are not enabled
session.disabled.cause = the server keeps no sessions; call Server.UseSessions before Start
session.key_missing.message = session key not found
session.key_missing.cause = the session holds no value under that key
session.invalid_entry = invalid session entry
session.invalid_entry.cause = a session key has 1 to 128 bytes and a value at most 4096 bytes
session.key_limit = session key limit exceeded
session.key_limit.cause = a session holds at most 64 keys

# Server sessions and handler queue (bn_core_web).
session.entropy_unavailable = session entropy provider unavailable
response.session_cookie_failed = session cookie failed
queue.closed = handler queue closed
server.listener_spawn_failed = failed to spawn listener thread
server.state_unavailable = server state unavailable
server.options_not_live = ServerOptions handle is not live
server.tls_config_not_live = BNWeb.TLSConfig handle is not live
server.invalid_address = invalid address
server.port_out_of_range = port out of range
server.options_non_negative = server option must be non-negative

# Contract names and type-mismatch contexts (bn_lib_web).
op.response.set_cookie = BNWeb.Response.SetCookie
op.response.set_cookie_with_policy = BNWeb.Response.SetCookieWithPolicy
op.response.clear_cookie = BNWeb.Response.ClearCookie
op.server.use_queue = BNWeb.Server.UseQueue
context.request.member_name = BNWeb.Request member name
context.cookie_policy = BOOLEAN, BOOLEAN, STRING
context.session.key_or_value = BNWeb.Session key or value
context.session.receiver = BNWeb.Session operation receiver
context.queue.not_live = Dispatch.Queue handle is not live
context.queue.non_queue = non-queue value
tls.ca_file_unreadable.cause = the CA file {path} cannot be read: {error}
tls.ca_file_empty.cause = the CA file {path} holds no valid certificate
tls.no_roots.cause = no trusted CA certificate is available: the system trust store holds none and no extra CA file was given
tls.trust_failed.message = TLS trust roots unavailable: {cause}
tls.platform_unreadable.cause = the system trust store cannot be read ({error}) and no extra CA file was given

# Transfers and client failures (bn_web web_error).
transfer.http_status.message = unexpected HTTP status {status}
transfer.http_status.cause = the server returned an HTTP status outside the 200..299 success range
transfer.already_exists.message = destination file "{path}" already exists
transfer.already_exists.cause = the destination file already exists and overwrite was not requested
transfer.unsupported_encoding.message = unsupported Content-Encoding "{encoding}"
transfer.unsupported_encoding.cause = the response body encoding is not supported (only gzip and deflate are supported)

# BNLog contract names and texts (language/0.6/bnlog.md, "Formats").
# One `key = value` per line; code refers to the key, never to the value.
# Every key here is used, and every key the code uses is here
# (bn_core_log `catalog` tests).

# Format names accepted by Logger.SetFormat.
format.json = JSON
format.syslog = SYSLOG
format.apache = APACHE

# What SYSLOG writes for an absent header field (RFC 5424 NILVALUE).
syslog.nil = -

# Fields an APACHE line reads, and what stands for an absent one.
apache.remote = http.remote
apache.method = http.method
apache.path = http.path
apache.protocol = http.protocol
apache.status = http.status
apache.bytes = http.bytes_sent
apache.referrer = http.referrer
apache.user_agent = http.user_agent
apache.duration_ms = http.duration_ms
apache.header_prefix = http.header.
apache.absent = -
apache.zone = +0000
apache.month = Jan Feb Mar Apr May Jun Jul Aug Sep Oct Nov Dec

# Sensitive keys: exact names, then markers a key may contain.
sensitive.exact = query body passwd tls_key
sensitive.marker = authorization cookie session password secret token api_key apikey access_key private_key credential bearer jwt signature

# Error for an unknown format ({format} is the text the program passed).
error.invalid_format.message = unknown log format "{format}"
error.invalid_format.cause = the format is "JSON", "SYSLOG" or "APACHE"



# What a bound names in an INVALID_ARGUMENT or OUT_OF_RANGE error.
what.label = logger label
what.destination = destination
what.field_value = field value
what.level = log level
what.timeout = timeout in milliseconds
error.apache_level.cause = an APACHE logger writes access records only (Logger.Access)

# Destinations named in Logger.New (bnlog.md, "Making a logger").
destination.console = CONSOLE
destination.null = NULL

# The fields of Log.Event, under their own names in a record.
event.service = service
event.request_id = request_id
event.route = route
event.user = user
event.detail = detail
event.code = code

# Contract names of the operations (Error.Operation).
op.logger.new = BNLog.Logger.New
op.logger.access = BNLog.Logger.Access
op.logger.set_level = BNLog.Logger.SetLevel
op.logger.set_defaults = BNLog.Logger.SetDefaults
op.logger.set_access_defaults = BNLog.Logger.SetAccessDefaults
op.logger.flush = BNLog.Logger.Flush
op.logger.close = BNLog.Logger.Close
op.logger.error = BNLog.Logger.Error
op.logger.warn = BNLog.Logger.Warn
op.logger.info = BNLog.Logger.Info
op.logger.http = BNLog.Logger.Http
op.logger.verbose = BNLog.Logger.Verbose
op.logger.debug = BNLog.Logger.Debug
op.logger.silly = BNLog.Logger.Silly

# Capabilities a destination needs (CAPABILITY_REQUIRED names them).
capability.console = HOST.Console
capability.filesystem = HOST.FileSystem
error.apache_level.message = cannot write a level record to an APACHE logger
error.capability.cause = the destination needs IMPORT {capability}
error.unavailable.cause = the BNLog member is not supplied by this runtime
error.released.message = BNLog.Logger was already deleted
context.non_struct = non-STRUCT value
